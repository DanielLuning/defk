defmodule App.EntropyTest do
  use ExUnit.Case, async: true
  use ExUnitProperties
  import Defk.Test, only: [assert_agrees: 3]

  defp unpack(bin), do: for(<<x::float-64-native <- bin>>, do: x)

  test "constant window has zero entropy" do
    assert unpack(App.Entropy.entropy(:binary.copy(<<7>>, 4096))) == [0.0]
  end

  test "uniform window has 8 bits" do
    w = for i <- 0..4095, into: <<>>, do: <<rem(i, 256)>>
    [h] = unpack(App.Entropy.entropy(w))
    assert_in_delta h, 8.0, 1.0e-12
  end

  test "partial window ignored, empty ok" do
    assert App.Entropy.entropy(<<1, 2, 3>>) == <<>>
    assert byte_size(App.Entropy.entropy(:binary.copy(<<1>>, 4096 * 2 + 100))) == 16
  end

  test "agrees with elixir mode on edges" do
    assert_agrees(App.Entropy, :entropy, [<<>>])
    assert_agrees(App.Entropy, :entropy, [:binary.copy(<<0>>, 4095)])
    assert_agrees(App.Entropy, :entropy, [:crypto.strong_rand_bytes(4096 * 3 + 5)])
  end

  property "agrees with elixir mode on generated inputs" do
    check all(bytes <- binary(min_length: 0, max_length: 9000), max_runs: 30) do
      assert_agrees(App.Entropy, :entropy, [bytes])
    end
  end
end
